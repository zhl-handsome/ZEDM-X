# ZDEM-X GPU 版本实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把 v8 per-vertex 罚弹簧 DEM 移植到 CUDA(`zdem_gpu`),FP64 与 CPU 逐步对照验证、FP32 过物理指标、64 粒子中等规模算例落地。

**Architecture:** 直写 CUDA kernel(SoA 粒子状态 + 体坐标网格注册表 + 静态墙组),空间哈希 broad-phase + CUB 排序,per-(pair,vertex) 并行罚弹簧 + atomicAdd 力累积,主循环序与 CPU 完全一致(力→积分→输出)。

**Tech Stack:** CUDA 12.8(CC 8.9 / RTX 4090 D)、CUB、CMake ≥3.18(CUDA language)、C++17、Python(对照脚本)。

**设计文档:** `docs/superpowers/specs/2026-08-17-gpu-dem-design.md`

## Global Constraints

- 精度宏:默认 `real = float`;CMake option `ZDEM_PRECISION=double` → `ZDEM_DOUBLE` → `real = double`。所有 device/host 物理计算用 `real`
- 接触物理公式与 CPU v8(`src/main.cpp`,commit 5c9595e)**逐条对应**,不改物理
- 每步 kernel 序 = CPU 循环序(清力 → 检测/力 → 积分 → 输出),无相位差
- `zdem_cpu` 保持可构建、回归通过(它是 golden reference)
- config 格式与 CPU 完全兼容;VTK 输出逐字段一致(分析脚本复用)
- FP64 验收:位置偏差 < 1e-6 m,能量相对偏差 < 1e-9(容差,非逐位)
- 环路遥测管线不移植;日志保留 contacts 计数
- 测试命令用 `D:/ProgramData/anaconda3/python.exe`;GPU exe 在 `build/Release/` 与 `build/Debug/` 均可构建,验证用 Release
- Windows + MSVC + nvcc:MSVC 主机编译器;`.cu` 文件里避免中文注释以外的问题(nvcc 对 UTF-8 无 BOM 源文件在 MSVC 下可能警告,统一 ASCII 注释)

  #pragma once
  #include <vector>
  #include <cuda_runtime.h>
  #include "gpu/real.hpp"
  #include "host/config_io.hpp"

  struct DeviceMeshes {          // flattened body-frame mesh registry
      // vertices: 3*total_v floats, tris: 3*total_t ints (vertex indices, GLOBAL)
      real* d_verts; int* d_tris;
      int* d_voffset;            // [n_mesh+1] vertex offset per mesh
      int* d_toffset;            // [n_mesh+1] tri offset per mesh
      real* d_mean_edge;         // [n_mesh]
      int n_mesh;
  };

  struct DeviceWalls {           // static world-space coplanar groups
      real* d_gn;                // [3*ng] group normal (push side = particle side)
      real* d_gd;                // [ng] plane offset: s = n.x + d, penetrating if s < -eps
      real* d_fp;                // [9*n_fp] footprints (3 verts each), row-major
      int* d_fp_start;           // [ng+1] footprint offset per group
      real* d_mu;                // [ng] friction (min particle/wall applied in kernel)
      int n_groups;
  };

  struct DeviceParticles {       // SoA state
      real *pos, *vel, *omega, *quat, *L;          // [3N],[3N],[3N],[4N],[3N]
      real *mass, *inv_mass, *radius, *equiv_radius;
      real *inertia_body_inv;                      // [9N] row-major body frame
      real *young, *poisson, *mu, *restitution;
      int *mesh_index;
      real *force, *torque;                        // [3N] zeroed each step
      int *contact_count;                          // [N]
      int n;
  };

  class GpuSim {
  public:
      DeviceParticles P; DeviceMeshes M; DeviceWalls W;
      real dt; real gravity[3]; real tangential_damping;
      void upload(const SimConfig& cfg);           // builds meshes/particles on host (same code path as CPU main), uploads
      void free_all();
      void download_soa(std::vector<double>& pos, std::vector<double>& vel,
                        std::vector<double>& omega) const;  // for output/verification
  };

- [ ] **Step 2: 实现 kernels**

  `integrate.cuh`(全文):
  ```cpp
  #pragma once
  #include "gpu/real.hpp"

  __global__ inline void clear_forces_kernel(real* force, real* torque, int* cc, int n) {
      int i = blockIdx.x * blockDim.x + threadIdx.x;
      if (i >= n) return;
      for (int k = 0; k < 3; ++k) { force[3*i+k] = 0; torque[3*i+k] = 0; }
      cc[i] = 0;
  }

  __device__ inline void quat_mul_device(const real a[4], const real b[4], real out[4]) {
      out[0] = a[0]*b[0] - a[1]*b[1] - a[2]*b[2] - a[3]*b[3];
      out[1] = a[0]*b[1] + a[1]*b[0] + a[2]*b[3] - a[3]*b[2];
      out[2] = a[0]*b[2] - a[1]*b[3] + a[2]*b[0] + a[3]*b[1];
      out[3] = a[0]*b[3] + a[1]*b[2] - a[2]*b[1] + a[3]*b[0];
  }
  __device__ inline void quat_rotate_device(const real q[4], const real v[3], real out[3]) {
      // v + 2*qv x (qv x v + w*v)  (matches CPU quat_rotate)
      real qv[3] = {q[1], q[2], q[3]};
      real c[3] = {qv[1]*v[2]-qv[2]*v[1], qv[2]*v[0]-qv[0]*v[2], qv[0]*v[1]-qv[1]*v[0]};
      real cc2[3] = {qv[1]*c[2]-qv[2]*c[1], qv[2]*c[0]-qv[0]*c[2], qv[0]*c[1]-qv[1]*c[0]};
      out[0] = v[0] + 2.0f*(q[0]*c[0] + cc2[0]);
      out[1] = v[1] + 2.0f*(q[0]*c[1] + cc2[1]);
      out[2] = v[2] + 2.0f*(q[0]*c[2] + cc2[2]);
  }
  __device__ inline void quat_step_world_device(const real q[4], const real w[3], real dt, real out[4]) {
      real wm = sqrtf(w[0]*w[0] + w[1]*w[1] + w[2]*w[2]);
      real th = wm * dt;
      real dq[4];
      if (th < real(1e-300)) { dq[0]=1; dq[1]=0; dq[2]=0; dq[3]=0; }
      else {
          real s = sinf(real(0.5)*th) / wm;
          dq[0] = cosf(real(0.5)*th);
          dq[1] = w[0]*s; dq[2] = w[1]*s; dq[3] = w[2]*s;
      }
      quat_mul_device(dq, q, out);
  }
  // math functions must switch on real: use overloads via <cmath> — in .cu use
  // sinf/cosf/sqrtf for float, sin/cos/sqrt for double. Provide wrappers:
  __device__ inline real r_sin(real x) { return sizeof(real)==4 ? sinf(x) : sin(x); }
  __device__ inline real r_cos(real x) { return sizeof(real)==4 ? cosf(x) : cos(x); }
  __device__ inline real r_sqrt(real x){ return sizeof(real)==4 ? sqrtf(x) : sqrt(x); }
  // (use r_* inside quat_step_world_device and later kernels; constexpr-if preferred:
  //   if constexpr (sizeof(real)==4) sinf else sin)

  __global__ inline void integrate_kernel(real* pos, real* vel, real* omega, real* quat, real* L,
                                          const real* force, const real* torque,
                                          const real* inv_mass, const real* inertia_body_inv,
                                          const real* gravity, real dt, int n) {
      int i = blockIdx.x * blockDim.x + threadIdx.x;
      if (i >= n) return;
      real acc[3];
      for (int k = 0; k < 3; ++k) {
          acc[k] = force[3*i+k] * inv_mass[i] + gravity[k];
          vel[3*i+k] += acc[k] * dt;
          pos[3*i+k] += vel[3*i+k] * dt;
      }
      for (int k = 0; k < 3; ++k) L[3*i+k] += torque[3*i+k] * dt;

      // world-frame I^-1 = R * Ibody^-1 * R^T  (R from quat)
      real q[4] = {quat[4*i], quat[4*i+1], quat[4*i+2], quat[4*i+3]};
      real R[9], Rt[9];
      // build rotation matrix from quat (same expansion as CPU mat3 from quat)
      real w2=q[0]*q[0], x2=q[1]*q[1], y2=q[2]*q[2], z2=q[3]*q[3];
      R[0]=w2+x2-y2-z2;  R[1]=2*(q[1]*q[2]-q[0]*q[3]); R[2]=2*(q[1]*q[3]+q[0]*q[2]);
      R[3]=2*(q[1]*q[2]+q[0]*q[3]); R[4]=w2-x2+y2-z2;  R[5]=2*(q[2]*q[3]-q[0]*q[1]);
      R[6]=2*(q[1]*q[3]-q[0]*q[2]); R[7]=2*(q[2]*q[3]+q[0]*q[1]); R[8]=w2-x2-y2+z2;
      for (int a = 0; a < 3; ++a) for (int b = 0; b < 3; ++b) Rt[3*b+a] = R[3*a+b];
      real Iinv[9];
      for (int a = 0; a < 3; ++a) for (int b = 0; b < 3; ++b) {
          real s = 0;
          for (int m2 = 0; m2 < 3; ++m2) for (int k2 = 0; k2 < 3; ++k2)
              s += R[3*a+m2] * inertia_body_inv[9*i+3*m2+k2] * Rt[3*k2+b];
          Iinv[3*a+b] = s;
      }
      auto IinvL = [&](const real Lin[3], real wout[3]) {
          for (int a = 0; a < 3; ++a) wout[a] = Iinv[3*a]*Lin[0] + Iinv[3*a+1]*Lin[1] + Iinv[3*a+2]*Lin[2];
      };
      real w0[3]; IinvL(&L[3*i], w0);
      real qh[4]; quat_step_world_device(q, w0, real(0.5)*dt, qh);
      // re-evaluate Iinv at half-step attitude
      // (repeat matrix build with qh; helper lambda build_Iinv(q, Iinv))
      real Iinv_h[9]; /* build from qh same as above */
      real wh[3]; real Lh[3] = {L[3*i], L[3*i+1], L[3*i+2]};
      for (int a = 0; a < 3; ++a) wh[a] = Iinv_h[3*a]*Lh[0] + Iinv_h[3*a+1]*Lh[1] + Iinv_h[3*a+2]*Lh[2];
      real q1[4]; quat_step_world_device(q, wh, dt, q1);
      for (int k = 0; k < 4; ++k) quat[4*i+k] = q1[k];
      real w1[3]; IinvL(&L[3*i], w1);   // final omega at NEW attitude (CPU does same)
      for (int k = 0; k < 3; ++k) omega[3*i+k] = w1[k];
  }
  ```
  **对照 CPU 语义逐行核对**(CPU 1080-1091:`w0` 用旧姿态 Iinv → 半步 → `wh` 用半步姿态 → 整步 → `omega` 用**新姿态** Iinv)。
